import cv2 import mediapipe as mp # Inicializar Mediapipe mp_hands = mp.solutions.hands mp_drawing = mp.solutions.drawing_utils hands = mp_hands.Hands(min_detection_confidence=0.7, min_tracking_confidence=0.7) # Inicializar cámara cap = cv2.VideoCapture(0) # Posición inicial de la figura circle_x, circle_y = 300, 300 while cap.isOpened(): success, image = cap.read() if not success: break # Convertir a RGB image_rgb = cv2.cvtColor(image, cv2.COLOR_BGR2RGB) results = hands.process(image_rgb) # Dibujar resultados if results.multi_hand_landmarks: for hand_landmarks in results.multi_hand_landmarks: mp_drawing.draw_landmarks(image, hand_landmarks, mp_hands.HAND_CONNECTIONS) # Obtener coordenadas del dedo índice (landmark 8) h, w, _ = image.shape index_x = int(hand_landmarks.landmark[8].x * w) index_y = int(hand_landmarks.landmark[8].y * h) # Mover el círculo a la posición del dedo índice circle_x, circle_y = index_x, index_y # Dibujar círculo cv2.circle(image, (circle_x, circle_y), 40, (0, 255, 0), -1) # Mostrar imagen cv2.imshow("Control con manos", image) if cv2.waitKey(1) & 0xFF == 27: # Presiona ESC para salir break cap.release() cv2.destroyAllWindows()
